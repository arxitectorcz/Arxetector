detailsLayout.Padding = UDim.new(0, 4)
detailsLayout.SortOrder = Enum.SortOrder.LayoutOrder
detailsLayout.Parent = detailsFrame

local function clearDetails()
    for _, child in pairs(detailsFrame:GetChildren()) do
        if child:IsA("TextLabel") then
            child:Destroy()
        end
    end
end

local function setVisibleRecursive(obj, visible)
    if obj:IsA("GuiObject") then
        obj.Visible = visible
    end
    for _, child in pairs(obj:GetChildren()) do
        setVisibleRecursive(child, visible)
    end
end

local function updateGUIList(query)
    for _, child in pairs(guiListFrame:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end

    local guis = {}
    for _, gui in pairs(playerGui:GetChildren()) do
        if gui:IsA("ScreenGui") and (query == "" or gui.Name:lower():find(query:lower())) then
            table.insert(guis, gui)
        end
    end

    table.sort(guis, function(a,b) return a.Name < b.Name end)

    for _, gui in ipairs(guis) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, 0, 0, 30)
        btn.BackgroundColor3 = gui.Enabled and Color3.fromRGB(0, 170, 0) or Color3.fromRGB(120, 120, 120)
        btn.TextColor3 = Color3.new(1,1,1)
        btn.Font = Enum.Font.SourceSans
        btn.TextSize = 18
        btn.Text = gui.Name
        btn.Parent = guiListFrame

        btn.MouseButton1Click:Connect(function()
            -- Переключаем Enabled
            gui.Enabled = not gui.Enabled

            -- Переключаем видимость рекурсивно
            setVisibleRecursive(gui, gui.Enabled)

            -- Обновляем цвет кнопки для отображения статуса
            btn.BackgroundColor3 = gui.Enabled and Color3.fromRGB(0, 170, 0) or Color3.fromRGB(120, 120, 120)

            -- Обновляем детали, если GUI теперь виден
            if gui.Enabled then
                clearDetails()
                showDetails(gui)
            else
                clearDetails()
            end
        end)
    end

    guiListFrame.CanvasSize = UDim2.new(0, 0, 0, guiListLayout.AbsoluteContentSize.Y)
end

function showDetails(gui)
    clearDetails()
    local function recurse(obj, indent)
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 0, 20)
        label.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
        label.TextColor3 = Color3.new(1, 1, 1)
        label.Font = Enum.Font.SourceSans
        label.TextSize = 16
        label.TextXAlignment = Enum.TextXAlignment.Left
        label.Text = string.rep("  ", indent) .. obj.Name .. " [" .. obj.ClassName .. "]"
        label.Parent = detailsFrame

        for _, child in pairs(obj:GetChildren()) do
            recurse(child, indent + 1)
        end
    end

    recurse(gui, 0)

    detailsFrame.CanvasSize = UDim2.new(0, 0, 0, detailsLayout.AbsoluteContentSize.Y)
end

searchBox:GetPropertyChangedSignal("Text"):Connect(function()
    updateGUIList(searchBox.Text)
end)

toggleButton.MouseButton1Click:Connect(function()
    mainFrame.Visible = not mainFrame.Visible
    toggleButton.Text = mainFrame.Visible and "-" or "+"
    if mainFrame.Visible then
        updateGUIList(searchBox.Text)
        clearDetails()
    end
end)

-- Изначально меню скрыто
mainFrame.Visible = false
toggleButton.Text = "+"
