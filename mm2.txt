-- === Вспомогательные функции и переменные ===
local function getRole(plr)
    if plr.Character then
        for _, tool in pairs(plr.Character:GetChildren()) do
            if tool:IsA("Tool") then
                local nameLower = tool.Name:lower()
                if nameLower:find("knife") then return "Murderer" end
                if nameLower:find("gun") then return "Sheriff" end
            end
        end
        local backpack = plr:FindFirstChildOfClass("Backpack")
        if backpack then
            for _, tool in pairs(backpack:GetChildren()) do
                if tool:IsA("Tool") then
                    local nameLower = tool.Name:lower()
                    if nameLower:find("knife") then return "Murderer" end
                    if nameLower:find("gun") then return "Sheriff" end
                end
            end
        end
    end
    return "Innocent"
end

-- Бессмертие
local invincibleConnection = nil
local invincibleEnabled = false

local function makeInvincible()
    local char = player.Character
    if not char then return end
    local humanoid = char:FindFirstChild("Humanoid")
    if not humanoid then return end

    if invincibleConnection then
        invincibleConnection:Disconnect()
        invincibleConnection = nil
    end

    invincibleConnection = humanoid.HealthChanged:Connect(function(newHealth)
        if invincibleEnabled and newHealth < humanoid.MaxHealth then
            humanoid.Health = humanoid.MaxHealth
        end
    end)
    invincibleEnabled = true
    print("Бессмертие включено")
end

local function disableInvincible()
    if invincibleConnection then
        invincibleConnection:Disconnect()
        invincibleConnection = nil
    end
    invincibleEnabled = false
    print("Бессмертие выключено")
end

local function toggleInvincible()
    if invincibleEnabled then
        disableInvincible()
    else
        makeInvincible()
    end
end

player.CharacterAdded:Connect(function()
    task.wait(0.1)
    if invincibleEnabled then
        makeInvincible()
    end
end)

-- Телепорт к игроку с ролью
local function teleportToRole(roleName)
    local function tryTeleport()
        for _, plr in pairs(Players:GetPlayers()) do
            if getRole(plr) == roleName and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    hrp.CFrame = plr.Character.HumanoidRootPart.CFrame + Vector3.new(2, 0, 0)
                    return true
                end
            end
        end
        return false
    end

    if not tryTeleport() then
        print(roleName .. " не найден или ты мертв, пробуем через 2 сек...")
        task.delay(2, function()
            if player.Character then
                tryTeleport()
            end
        end)
    end
end

local function tpToMurderer()
    teleportToRole("Murderer")
end

local function tpToSheriff()
    teleportToRole("Sheriff")
end

local function tpToRandomPlayer()
    local alivePlayers = {}
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
            table.insert(alivePlayers, plr)
        end
    end

    if #alivePlayers == 0 then
        print("Нет живых игроков для телепорта")
        return
    end

    local randomPlayer = alivePlayers[math.random(1, #alivePlayers)]
    local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if hrp and randomPlayer.Character and randomPlayer.Character:FindFirstChild("HumanoidRootPart") then
        hrp.CFrame = randomPlayer.Character.HumanoidRootPart.CFrame + Vector3.new(2, 0, 0)
        print("Телепортировался к случайному игроку: "..randomPlayer.Name)
    else
        print("Не удалось телепортироваться")
    end
end

-- Noclip
local noclipConn
local noclipEnabled = false

local function startNoclip()
    if noclipConn then return end
    noclipConn = RunService.Heartbeat:Connect(function()
        local character = player.Character
        if not character then return end
        for _, part in pairs(character:GetChildren()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                part.CanCollide = false
            end
        end
    end)
end

local function stopNoclip()
    if noclipConn then
        noclipConn:Disconnect()
        noclipConn = nil
    end
    local character = player.Character
    if character then
        for _, part in pairs(character:GetChildren()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                part.CanCollide = true
            end
        end
    end
end

local function toggleNoclip()
    noclipEnabled = not noclipEnabled
    if noclipEnabled then
        startNoclip()
        print("🚫 Noclip включён")
    else
        stopNoclip()
        print("✅ Noclip выключен")
    end
end

-- Infinite jump
local infiniteJumpEnabled = false
local jumpConn = nil

local function toggleInfiniteJump()
    infiniteJumpEnabled = not infiniteJumpEnabled
    if infiniteJumpEnabled then
        if not jumpConn then
            jumpConn = UserInputService.JumpRequest:Connect(function()
                if infiniteJumpEnabled and player.Character and player.Character:FindFirstChild("Humanoid") then
                    player.Character.Humanoid:ChangeState("Jumping")
                end
            end)
        end
        print("🦘 Бесконечный прыжок включён")
    else
        if jumpConn then
            jumpConn:Disconnect()
            jumpConn = nil
        end
        print("🦘 Бесконечный прыжок выключен")
    end
end

-- Speed boost

        humanoid.WalkSpeed = speedEnabled and boostedSpeed or normalSpeed
        print("🏃 Скорость " .. (speedEnabled and "x2" or "x1"))
    end
end
