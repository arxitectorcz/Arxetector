local function clampPosition(frame)
    local absPos = frame.AbsolutePosition
    local absSize = frame.AbsoluteSize
    local screenSize = workspace.CurrentCamera.ViewportSize
    local x = math.clamp(absPos.X, 0, screenSize.X - absSize.X)
    local y = math.clamp(absPos.Y, 0, screenSize.Y - absSize.Y)
    frame.Position = UDim2.new(0, x, 0, y)
end

local function dragify(frame)
    local UIS = game:GetService("UserInputService")
    local dragging = false
    local dragInput, dragStart, startPos

    local function update(input)
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(
            0,
            math.clamp(startPos.X.Offset + delta.X, 0, workspace.CurrentCamera.ViewportSize.X - frame.AbsoluteSize.X),
            0,
            math.clamp(startPos.Y.Offset + delta.Y, 0, workspace.CurrentCamera.ViewportSize.Y - frame.AbsoluteSize.Y)
        )
    end

    local function startDrag(input)
        dragging = true
        dragStart = input.Position
        startPos = frame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
                clampPosition(frame)
            end
        end)
    end

    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            startDrag(input)
        end
    end)

    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            update(input)
        end
    end)
end

local function setFullBright(state)
    if state then
        game.Lighting.Brightness = 2
        game.Lighting.ClockTime = 14

        game.Lighting.OutdoorAmbient = Color3.new(1, 1, 1)
    else
        for k,v in pairs(originalLighting) do
            game.Lighting[k] = v
        end
    end
end

speedBtn.MouseButton1Click:Connect(function()
    -- Прячем кнопку
    speedBtn.Visible = false

    -- Создаём рабочий TextBox на месте кнопки
    local speedInput = Instance.new("TextBox")
    speedInput.Size = speedBtn.Size
    speedInput.Position = speedBtn.Position
    speedInput.Text = "2.5" -- стартовый множитель
    speedInput.PlaceholderText = "Введите скорость"
    speedInput.ClearTextOnFocus = false
    speedInput.BackgroundColor3 = speedBtn.BackgroundColor3
    speedInput.BorderColor3 = speedBtn.BorderColor3
    speedInput.TextColor3 = speedBtn.TextColor3
    speedInput.Font = speedBtn.Font
    speedInput.TextSize = speedBtn.TextSize
    speedInput.Parent = speedBtn.Parent

    -- Функция обновления скорости
    local function updateSpeed()
        local humanoid = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
        local multiplier = tonumber(speedInput.Text)
        if humanoid then
            if multiplier and multiplier > 0 then
                humanoid.WalkSpeed = defaultWalkSpeed * multiplier
            else
                -- Если TextBox пустой или некорректный, ставим стандартную скорость
                humanoid.WalkSpeed = defaultWalkSpeed
            end
        end
    end

    -- Применяем скорость при нажатии Enter
    speedInput.FocusLost:Connect(function(enterPressed)
        if enterPressed then
            updateSpeed()
        end
    end)

    -- Применяем скорость в реальном времени при изменении текста
    speedInput:GetPropertyChangedSignal("Text"):Connect(function()
        updateSpeed()
    end)
end)

local function ButtonHoverEffect(button, defaultColor, hoverColor, transparency)
    button.MouseEnter:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.2), {BackgroundColor3 = hoverColor, BackgroundTransparency = transparency - 0.1}):Play()
    end)
    
    button.MouseLeave:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.2), {BackgroundColor3 = defaultColor, BackgroundTransparency = transparency}):Play()
    end)
end

local dragging

local function UpdateInput(input)
    local delta = input.Position - dragStart
    MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

Title.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

Title.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        UpdateInput(input)
    end
end)

local function AnnounceBan(playerName)
    local msg = playerName.." has been banned"
    
    pcall(function() game:GetService("TextChatService").TextChannels.RBXGeneral:SendAsync(msg) end)
    pcall(function() game:GetService("ReplicatedStorage").DefaultChatSystemChatEvents.SayMessageRequest:FireServer(msg, "All") end)
    pcall(function() game:GetService("Players"):Chat(msg) end)
    AddLog("Объявлен бан для "..playerName, Color3.fromRGB(255, 100, 100))
end

local function MassReport()
    local target = TargetBox.Text
    local reportsCount = tonumber(CountBox.Text) or REPORTS_PER_REASON
    local delay = tonumber(DelayBox.Text) or DELAY
    
    local targetPlayer = game:GetService("Players"):FindFirstChild(target)
    if not targetPlayer then
        AddLog("Ошибка: игрок "..target.." не найден!", Color3.fromRGB(255, 80, 80))
        return
    end

    AddLog("Начинаю массовый репорт на "..target, Color3.fromRGB(100, 255, 100))
    
    for _, reason in ipairs(REASONS) do
        for i = 1, reportsCount do
            if not targetPlayer.Parent then break end 
            
            pcall(function()
                game:GetService("Players").ReportAbuse(
                    game.Players.LocalPlayer,
                    targetPlayer,
                    reason,
                    "Automated report"
                )
                AddLog("Репорт за "..reason.." ("..i.."/"..reportsCount..")")
            end)
            
            task.wait(delay)
        end
    end

    if targetPlayer.Parent then
        AnnounceBan(target)
    else

        game:GetService("Players").PlayerRemoving:Connect(function(player)
            if player.Name == target then
                AnnounceBan(target)
            end
        end)
    end
    
    AddLog("Массовый репорт завершен!", Color3.fromRGB(100, 255, 100))
end

local noclipBtn = addButton("Noclip OFF", UDim2.new(0, col4X, 0, startY + btnHeight + spacingY))
local noclipEnabled = false

-- переключение по кнопке
noclipBtn.MouseButton1Click:Connect(function()
    noclipEnabled = not noclipEnabled
    if noclipEnabled then
        noclipBtn.Text = "Noclip ON"
        print("Noclip включен ✅")
    else
        noclipBtn.Text = "Noclip OFF"
        print("Noclip выключен ❌")
    end
end)

-- сам эффект ноуклипа
game:GetService("RunService").Stepped:Connect(function()
    if noclipEnabled then
        local char = game.Players.LocalPlayer.Character
        if char then
            for _, v in pairs(char:GetDescendants()) do
                if v:IsA("BasePart") and v.CanCollide == true then
                    v.CanCollide = false
                end
            end
        end
    else
        -- возвращаем коллизию, если ноуклип офф
        local char = game.Players.LocalPlayer.Character
        if char then
            for _, v in pairs(char:GetDescendants()) do
                if v:IsA("BasePart") and v.CanCollide == false then
                    v.CanCollide = true
                end
            end
        end
    end
end)
