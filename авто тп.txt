title.Parent = mainFrame

-- Драггинг меню
local dragging = false
local dragOffset
title.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragOffset = mainFrame.Position - UDim2.new(0, input.Position.X, 0, input.Position.Y)
    end
end)
title.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)
title.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement and dragging then
        mainFrame.Position = UDim2.new(0, input.Position.X + dragOffset.X.Offset, 0, input.Position.Y + dragOffset.Y.Offset)
    end
end)

UIListLayout.Padding = UDim.new(0,5)

local tpList = {}
local autoTPActive = false
local autoTPCounter = 0

local function refreshTPList()
    listFrame:ClearAllChildren()
    for i, pos in ipairs(tpList) do
        local tpButton = Instance.new("TextButton")
        tpButton.Size = UDim2.new(1,0,0,25)
        tpButton.BackgroundColor3 = Color3.fromRGB(60,60,60)
        tpButton.TextColor3 = Color3.fromRGB(255,255,255)
        tpButton.Text = string.format("TP %d: X=%.1f Y=%.1f Z=%.1f", i, pos.X, pos.Y, pos.Z)
        tpButton.Parent = listFrame

        local lastClick = 0
        tpButton.MouseButton1Click:Connect(function()
            local now = tick()
            if now - lastClick < 0.4 then
                table.remove(tpList, i)
                refreshTPList()
            else
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    player.Character.HumanoidRootPart.CFrame = CFrame.new(pos)
                end
            end
            lastClick = now
        end)
    end
end

addTPButton.MouseButton1Click:Connect(function()
    local position = player.Character and player.Character:FindFirstChild("HumanoidRootPart") and player.Character.HumanoidRootPart.Position
    if position then
        table.insert(tpList, position)
        refreshTPList()
    end
end)

spawn(function()
    while true do
        wait()
        if autoTPActive and #tpList > 0 then
            for i, pos in ipairs(tpList) do
                if not autoTPActive then break end
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    player.Character.HumanoidRootPart.CFrame = CFrame.new(pos)
                    wait(0.5)
                end
            end
        end
    end
end)

autoTPButton.MouseButton1Click:Connect(function()
    if not autoTPActive then
        autoTPActive = true
        autoTPCounter = 1
    else
        autoTPCounter = autoTPCounter + 1
        if autoTPCounter >= 2 then
            autoTPActive = false
            autoTPCounter = 0
        end
    end
end)
